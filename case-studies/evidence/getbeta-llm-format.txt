def validate_provider_response_format(
    response_format: object,
    *,
    supported: bool,
) -> None:
    """Reject malformed or unsupported structured-output requests fail-closed."""
    if response_format is None:
        return
    if type(response_format) is not dict:
        raise ValueError("provider response_format must be an object")
    if not supported:
        raise ValueError("provider does not support response_format")
