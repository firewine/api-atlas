def provider_operation_deadline(timeout_s: float) -> float:
    """Return the monotonic deadline shared by one logical operation."""
    return time.monotonic() + timeout_s


def remaining_provider_timeout(deadline: float) -> float:
    """Return remaining operation time or fail before another SDK call."""
    remaining = deadline - time.monotonic()
    if remaining <= 0:
        raise TimeoutError("LLM provider deadline exceeded")
    return min(remaining, PROVIDER_TIMEOUT_SECONDS)
